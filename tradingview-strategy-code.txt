// Fuction to remove parts of the ticker
f_remove_exchange_name(string name) =>
    string[] parts      = str.split(name, ":")
    string result       = array.size(parts) > 1 ? array.get(parts, 1) : name
    result

// Function to get color based on the value of the signal
f_trend_col(series int x) =>
    x > 0 ? color.green : color.red

// Gaussian filter (source weighted by a Gaussian kernel) - used by the UNISTRAT Gaussian VIDYA
f_gaussian_filter(series float src, simple int length, simple float sigma) =>
    gaussian_sum          = 0.0
    gaussian_weighted_sum = 0.0
    for i = 0 to length - 1 by 1
        weight                = math.exp(-0.5 * math.pow((i - (length - 1) / 2) / sigma, 2))
        gaussian_sum          += weight
        gaussian_weighted_sum += src[i] * weight
    gaussian_weighted_sum / gaussian_sum

// For-loop scoring of a moving average (counts how often the current value is above past values) - used by the UNISTRAT VIDYA Loop
f_loop(simple int a, simple int b, series float ma) =>
    sum2 = 0.0
    for i = a to b by 1
        sum2 += (ma > ma[i] ? 1 : -1)
    sum2

// Function to return opposite of the signal
f_oppose(series int x) =>
    x == 1 ? (0) : (x == 0 ? 1 : 0)

// Function to calculate the age of the ticker
f_age() =>
    // Calculate number of bars since the beginning of the chart
    bars = bar_index + 1
    // Limit the age to 5000 if it exceeds
    math.min(bars, 5000)

// Function to return the daily return as decimal number
f_roc() =>
    roc = ta.roc(close, 1)
    r   = roc/100

// Function to find the strongest, second-strongest AND third-strongest tickers among 7
// (ticker, value) pairs. Returns [best, second, third]. All are derived from the same
// ranking so they stay consistent.
// TIE-BREAK: when two entities have the same score, the winner is decided by their DIRECT
// pairwise UNISTRAT classification (pw.get(i, j) == 1 means entity i beats entity j in the
// i/j ratio), so results don't depend on the order the tickers are entered in.
f_TopThreeAssets(simple string ticker1, series int v1, simple string ticker2, series int v2, simple string ticker3, series int v3, 
 simple string ticker4, series int v4, simple string ticker5, series int v5, simple string ticker6, series int v6, simple string ticker7, series int v7, matrix<int> pw) =>
    // Create arrays for tickers and values
    var array<string> tickers = array.new_string(7, "")
    var array<float> values   = array.new_float(7, 0.0)
    
    array.set(tickers, 0, ticker1), array.set(values, 0, v1)
    array.set(tickers, 1, ticker2), array.set(values, 1, v2)
    array.set(tickers, 2, ticker3), array.set(values, 2, v3)
    array.set(tickers, 3, ticker4), array.set(values, 3, v4)
    array.set(tickers, 4, ticker5), array.set(values, 4, v5)
    array.set(tickers, 5, ticker6), array.set(values, 5, v6)
    array.set(tickers, 6, ticker7), array.set(values, 6, v7)
    
    // Index of the highest value; on ties, the challenger takes the lead only if it beats
    // the current leader in their direct pairwise ratio classification
    int bestIndex = 0
    for i = 1 to 6
        float v_i = array.get(values, i)
        float v_b = array.get(values, bestIndex)
        if v_i > v_b or (v_i == v_b and matrix.get(pw, i, bestIndex) == 1)
            bestIndex := i

    // Index of the highest value among the remaining (the second-strongest), same tie-break
    int secondIndex = bestIndex == 0 ? 1 : 0
    for i = 0 to 6
        if i != bestIndex
            float v_i = array.get(values, i)
            float v_s = array.get(values, secondIndex)
            if v_i > v_s or (v_i == v_s and matrix.get(pw, i, secondIndex) == 1)
                secondIndex := i

    // Index of the highest value among the rest (the third-strongest), same tie-break.
    // Seed with the lowest index not already taken by best/second.
    int thirdIndex = bestIndex != 0 and secondIndex != 0 ? 0 : bestIndex != 1 and secondIndex != 1 ? 1 : 2
    for i = 0 to 6
        if i != bestIndex and i != secondIndex
            float v_i = array.get(values, i)
            float v_t = array.get(values, thirdIndex)
            if v_i > v_t or (v_i == v_t and matrix.get(pw, i, thirdIndex) == 1)
                thirdIndex := i

    [array.get(tickers, bestIndex), array.get(tickers, secondIndex), array.get(tickers, thirdIndex)]

// Function to calculate strategy equity, with maximum drawdown.
// Allocation is driven by the weights: (1,0,0) -> 100% strongest; (0.8,0.2,0) -> 80/20
// strongest/second; (0.6,0.3,0.1) -> 60/30/10 across the top three. Blends are rebalanced
// every bar, so the bar's portfolio return is the weighted sum of the legs' returns
// (decided on the prior close via the [1] index, as before).
f_equity(simple string ticker1, simple string ticker2, simple string ticker3, simple string ticker4, simple string ticker5, simple string ticker6,  
 series string best_asset, series string second_asset, series string third_asset, series float r1, series float r2, series float r3, series float r4, series float r5, series float r6, 
 series bool backtest, simple float w1, simple float w2, simple float w3, series bool btc_ok, series float cash_ret) =>

    var float equity        = na
    var float peak_equity   = na
    var float max_drawdown  = 0.0

    if backtest
        // "USD" legs earn the cash-sleeve return: 0 normally, or trend-managed gold's
        // return when the gold-while-in-cash option is enabled and gold's weekly trend is long.
        a_best = switch best_asset[1]
            ticker1 => r1
            ticker2 => r2
            ticker3 => r3
            ticker4 => r4
            ticker5 => r5
            ticker6 => r6
            "USD"   => cash_ret

        a_second = switch second_asset[1]
            ticker1 => r1
            ticker2 => r2
            ticker3 => r3
            ticker4 => r4
            ticker5 => r5
            ticker6 => r6
            "USD"   => cash_ret

        a_third = switch third_asset[1]
            ticker1 => r1
            ticker2 => r2
            ticker3 => r3
            ticker4 => r4
            ticker5 => r5
            ticker6 => r6
            "USD"   => cash_ret

        // When USD is the strongest asset the system is fully risk-off, so we hold 100%
        // cash (the cash sleeve) regardless of mode (no extra legs into weaker assets).
        best_is_usd = best_asset[1] == "USD"
        // Market-regime filter: only take a position while BTC is in an uptrend (per the BTC
        // filters), else hold cash. Evaluated on the prior bar ([1]) to match best_asset[1] timing.
        btc_allows  = btc_ok[1]
        // Blended (rebalanced) return for the bar (cash sleeve when a BTC filter blocks)
        a = not btc_allows ? cash_ret :
             best_is_usd ? a_best :
             w2 == 0.0 and w3 == 0.0 ? a_best :
             w3 == 0.0 ? w1 * a_best + w2 * a_second :
             w1 * a_best + w2 * a_second + w3 * a_third

        equity              := na(equity[1]) ? 1 : equity[1] * (1 + a)

        // Calculate max drawdown
        peak_equity         := math.max(na(peak_equity) ? equity : peak_equity, equity)
        current_drawdown    = peak_equity > 0 ? (peak_equity - equity) / peak_equity : 0
        if current_drawdown > max_drawdown
            max_drawdown    := current_drawdown
        if equity > peak_equity
            peak_equity     := equity
    else
        equity              := 1

    [equity, math.round(max_drawdown, 4)]

// Function to calculate buy and hold equity of an asset, with maximum drawdown
f_buy_and_hold(series float r, series bool backtest) =>

    var float equity        = na
    var float peak_equity   = na
    var float max_drawdown  = 0.0
    
    if backtest
        // Calculate equity
        equity              := na(equity[1]) ? 1 : equity[1] * (1 + r)
        
        // Calculate max drawdown
        peak_equity         := math.max(na(peak_equity) ? equity : peak_equity, equity)
        current_drawdown    = peak_equity > 0 ? (peak_equity - equity) / peak_equity : 0
        if current_drawdown > max_drawdown
            max_drawdown    := current_drawdown
        if equity > peak_equity
            peak_equity     := equity
    else
        equity              := 1

    [equity, math.round(max_drawdown, 4)]

// Function to calculate the "filtered" equity, with maximum drawdown. It compounds the
// unfiltered strategy equity's return ONLY while that equity is itself in a positive
// UNISTRAT trend (eq_signal > 0); otherwise it sits flat (in cash). eq_signal is the
// UNISTRAT long signal computed on the unfiltered equity curve, and it is referenced
// with a one-bar lag so the in/out decision is made on the prior close (no look-ahead).
f_filtered_equity(series float base_equity, series int eq_signal, series bool backtest, series float cash_ret) =>

    var float equity        = na
    var float peak_equity   = na
    var float max_drawdown  = 0.0

    if backtest
        // Invested only if the equity was trending up as of the previous bar
        invested            = eq_signal[1] > 0
        // The unfiltered equity's bar return = the return of the asset held this bar
        eq_ret              = nz(base_equity / base_equity[1] - 1, 0)
        // While filtered out, the book sits in the cash sleeve (0, or trend-managed gold)
        a                   = invested ? eq_ret : cash_ret

        equity              := na(equity[1]) ? 1 : equity[1] * (1 + a)

        // Calculate max drawdown
        peak_equity         := math.max(na(peak_equity) ? equity : peak_equity, equity)
        current_drawdown    = peak_equity > 0 ? (peak_equity - equity) / peak_equity : 0
        if current_drawdown > max_drawdown
            max_drawdown    := current_drawdown
        if equity > peak_equity
            peak_equity     := equity
    else
        equity              := 1

    [equity, math.round(max_drawdown, 4)]

// Function to calculate performance metrics from equity curve ("base" - is the equity curve)
f_PerformanceMetrics(series float base, series int Lookback, series bool backtest, series float max_drawdown) =>
    // Initialize variables for positive and negative returns
    pos_sum                     = 0.0
    neg_sum                     = 0.0
    pos_count                   = 0
    neg_count                   = 0
    returns_sum                 = 0.0
    returns_squared_sum         = 0.0
    pos_returns_squared_sum     = 0.0
    neg_returns_squared_sum     = 0.0

    lookback                    = Lookback

    if Lookback > 4999
        lookback := 4998

    // Loop through the past 'Lookback' bars to calculate sums, counts, and drawdown
    if backtest
        for i = 0 to lookback - 1
            r = (base[i] - base[i + 1]) / base[i + 1]
            returns_sum += r
            returns_squared_sum += r * r

            if r > 0
                pos_sum   += r
                pos_count += 1
                pos_returns_squared_sum += r * r

            else if r < 0
                neg_sum   += r
                neg_count += 1
                neg_returns_squared_sum += r * r
    
    float[] export_array = array.new_float(10)

    // Calculate means
    mean_all        = math.round(returns_sum / lookback, 4)
    mean_pos        = pos_count != 0 ? math.round(pos_sum / pos_count, 4) : na
    mean_neg        = neg_count != 0 ? math.round(neg_sum / neg_count, 4) : na

    // Calculate standard deviations
    stddev_all      = math.round(math.sqrt((returns_squared_sum - (returns_sum * returns_sum) / lookback) / lookback), 4)
    stddev_pos      = pos_count != 0 ? math.round(math.sqrt((pos_returns_squared_sum - (pos_sum * pos_sum) / pos_count) / pos_count), 4) : na
    stddev_neg      = neg_count != 0 ? math.round(math.sqrt((neg_returns_squared_sum - (neg_sum * neg_sum) / neg_count) / neg_count), 4) : na

    // Calculate ratios. Annualization is derived from the actual chart timeframe assuming a
    // 24/7 market (365 days): bars per year = seconds per year / seconds per bar. On a 12h
    // chart this is sqrt(730) - the old hardcoded sqrt(255) understated Sharpe/Sortino by ~40%.
    annualization   = math.sqrt(31536000.0 / timeframe.in_seconds(timeframe.period))
    sharpe_ratio    = math.round((mean_all / stddev_all) * annualization, 3)
    sortino_ratio   = math.round((mean_all / stddev_neg) * annualization, 3)
    omega_ratio     = math.round(pos_sum / math.abs(neg_sum), 3)

    // Set values in the array
    array.set(export_array, 0, mean_all     * 100)
    array.set(export_array, 1, mean_pos     * 100)
    array.set(export_array, 2, mean_neg     * 100)
    array.set(export_array, 3, stddev_all   * 100)
    array.set(export_array, 4, stddev_pos   * 100)
    array.set(export_array, 5, stddev_neg   * 100)
    array.set(export_array, 6, sharpe_ratio)
    array.set(export_array, 7, sortino_ratio)
    array.set(export_array, 8, omega_ratio)
    array.set(export_array, 9, max_drawdown * 100)

    // Export the array
    export_array

// Function to plot Perfomance Metrics Table

    // Header (merged across all used columns: labels + Strategy + the included assets)
    table.cell(p, 0, 0, "Performance Metrics", text_color = color.white, bgcolor = color.gray)
    table.merge_cells(p, 0, 0, num_assets + 1, 0)

    // Column headers (only the included assets)
    table.cell(p, 1, 1, "Strategy", text_color = array.get(colors, 7))
    if num_assets >= 1
        table.cell(p, 2, 1, str.tostring(f_remove_exchange_name(ticker1)), text_color = array.get(colors, 0))
    if num_assets >= 2
        table.cell(p, 3, 1, str.tostring(f_remove_exchange_name(ticker2)), text_color = array.get(colors, 1))
    if num_assets >= 3
        table.cell(p, 4, 1, str.tostring(f_remove_exchange_name(ticker3)), text_color = array.get(colors, 2))
    if num_assets >= 4
        table.cell(p, 5, 1, str.tostring(f_remove_exchange_name(ticker4)), text_color = array.get(colors, 3))
    if num_assets >= 5
        table.cell(p, 6, 1, str.tostring(f_remove_exchange_name(ticker5)), text_color = array.get(colors, 4))
    if num_assets >= 6
        table.cell(p, 7, 1, str.tostring(f_remove_exchange_name(ticker6)), text_color = array.get(colors, 5))

    // Metrics labels
    for i = 0 to array.size(strategy) - 1

        metric = switch i
            0  => "Mean (All Returns) %"
            1  => "Mean (Positive Returns) %"
            2  => "Mean (Negative Returns) %"

            table.cell(p, 4, i + 2, str.tostring(array.get(a3, i)),     text_color = array.get(colors, 2))

        if num_assets >= 5
            table.cell(p, 6, i + 2, str.tostring(array.get(a5, i)),     text_color = array.get(colors, 4))
        if num_assets >= 6
            table.cell(p, 7, i + 2, str.tostring(array.get(a6, i)),     text_color = array.get(colors, 5))
    p

// ╔════════════════════════════╗
// ║   TREND CLASSIFICATION     ║
// ║   (UNISTRAT TPI)           ║
// ╚════════════════════════════╝
//{
// Replaces the old single-indicator trend following classifiers with the UNISTRAT
// Trend Probability Index. Four independent sub-models each return a vote in {-1, 0, +1}:
//   1) Gaussian VIDYA          -> B1
//   2) DEMA ATR                -> DemaAtrScore
//   3) VIDYA For-Loop          -> QB2
//   4) SD Median               -> sdMedianScore
// The votes are averaged into tpiScore. A positive TPI means the (numerator) asset of the
// evaluated pair is in a relative uptrend, i.e. it is outperforming the (denominator) asset.
//
// This function is defined after the INPUTS section so it can read the UNISTRAT settings
// directly from global scope. The classification "src" is passed in as an argument so that,
// when this function is called inside request.security(), it is evaluated in the context of
// the requested (ratio) symbol - exactly how the original script handled its sources.
// Core UNISTRAT TPI. `synthetic` must be true when `src` is a constructed series with no
// real OHLC (e.g. the equity curve): in that case the ATR sub-models cannot use ta.atr()
// (which reads the chart's high/low/close), so the true range collapses to the absolute
// bar-to-bar change of src. For real symbols (the asset-ratio charts) pass synthetic=false
// so ta.atr() uses the requested symbol's own OHLC, exactly as before.
f_tpi(series float src, simple bool synthetic) =>

    // ----- 1) Gaussian VIDYA -----
    gaussian_smooth = f_gaussian_filter(src, g_periode, g_sigma)
    k_g             = ta.stdev(gaussian_smooth, v_per) / ta.stdev(gaussian_smooth, v_histPer)
    sc_g            = 2.0 / (v_per + 1)
    var float VIDYA = 0.0
    VIDYA           := nz(k_g * sc_g * gaussian_smooth + (1 - k_g * sc_g) * VIDYA[1], 0)
    atr_g           = synthetic ? ta.rma(nz(math.abs(src - src[1])), v_atr_length) : ta.atr(v_atr_length)
    long_g          = src > VIDYA + atr_g * v_mult
    short_g         = src < VIDYA - atr_g * v_mult
    var int B1      = 0
    if long_g and not short_g
        B1 := 1
    if short_g
        B1 := -1

    // ----- 2) DEMA ATR -----
    GDEMA               = ta.dema(src, d_periodDema)
    atr_d               = synthetic ? ta.rma(nz(math.abs(src - src[1])), d_periodAtr) : ta.atr(d_periodAtr)
    long_d              = src > GDEMA + atr_d * d_factorAtr
    short_d             = src < GDEMA - atr_d * d_factorAtr
    var int DemaAtrScore = 0
    if long_d and not short_d
        DemaAtrScore := 1
    if short_d
        DemaAtrScore := -1

    // ----- 3) VIDYA For-Loop -----
    SD_V             = ta.stdev(src, l_len_V) / ta.stdev(src, l_len_VHist)
    sc_l             = 2.0 / (l_len_V + 1)
    var float VIDYA2 = 0.0
    VIDYA2           := nz(SD_V * sc_l * src + (1 - SD_V * sc_l) * VIDYA2[1], 0)
    vidya_loop       = f_loop(l_start, l_end, VIDYA2)
    long_l           = vidya_loop > l_long_t
    short_l          = vidya_loop < l_short_t
    var int QB2      = 0
    if long_l and not short_l
        QB2 := 1
    if short_l
        QB2 := -1

    // ----- 4) SD Median -----
    // UNISTRAT's median model is written against `close`; here it uses `src` so the
    // whole TPI can be evaluated on ANY series - the ratio's price when classifying
    // asset pairs (src resolves to the requested symbol's close by default, so this
    // is unchanged for the default source) and the equity curve when the equity
    // filter is enabled.
    base_median         = ta.median(src, m_base_len)
    base_medianSd       = ta.stdev(base_median, m_sd_len)
    upperBasemedian     = (base_median + base_medianSd) * m_upperSd
    lowerBasemedian     = (base_median - base_medianSd) * m_lowerSd
    base_long           = src > upperBasemedian
    base_short          = src < lowerBasemedian
    normBasemedian      = ta.median(src, m_norm_len)
    normMedian          = -1 * normBasemedian / src
    normSd              = ta.stdev(normMedian, m_sd_len)
    normLowerBound      = ta.ema(normMedian - normSd, m_normSmooth)
    norm_long           = normLowerBound > -1
    norm_short          = normMedian < -1
    long_m              = base_long and norm_long
    short_m             = base_short and norm_short
    var int sdMedianScore = 0
    if long_m and not short_m
        sdMedianScore := 1
    if short_m
        sdMedianScore := -1

    // ----- Combine into the TPI -----
    // tpiScore  > 0  -> numerator asset wins (long)
    // tpiScore  < 0  -> denominator asset wins (flat)
    // tpiScore == 0  -> hold the previous signal. This mirrors how UNISTRAT was
    //                   designed: it only flips the position when the score crosses
    //                   zero (long on > 0, cash on < 0), leaving the position
    //                   untouched while the score sits exactly at zero.
    tpiScore    = math.avg(B1, DemaAtrScore, QB2, sdMedianScore)
    var int sig = 0
    if tpiScore > 0
        sig := 1
    if tpiScore < 0
        sig := 0

    [sig, f_oppose(sig)]

// Thin wrapper used for the asset-pair classification (real symbols -> synthetic = false),
// so all the request.security() call sites can keep calling f_trend(src) unchanged.
f_trend(series float src) =>
    f_tpi(src, false)

// Non-repainting wrapper for HIGHER-TIMEFRAME requests. Returns the signal of the LAST
// CLOSED higher-timeframe bar. Must be paired with lookahead = barmerge.lookahead_on in
// request.security() - the standard non-repainting HTF pattern: the [1] offset means only
// confirmed HTF bars are ever referenced, and lookahead_on aligns that confirmed value to
// the chart bars where it was actually known. Result: historical and realtime bars see
// EXACTLY the same series - no more weekly signals that appear mid-week and vanish.
f_trend_htf(series float src) =>
    [s, o] = f_tpi(src, false)
    [s[1], o[1]]

// ════════════════════════════════════════════════════════════════════════════
//  WEEKLY EQUITY EXPORT (OHLC)  ->  Pine Logs panel   (works on the 12h chart)
//  One "YYYY-MM-DD <tab> open <tab> high <tab> low <tab> close" per week-end,
//  oldest first, plus the current (partial) week on the latest bar.
//  Value = followed equity (total portfolio value, incl. cash sleeve).
//    open  = equity at the FIRST chart bar of the week
//    high  = highest equity recorded during the week
//    low   = lowest  equity recorded during the week
//    close = equity at the LAST chart bar of the week
//  Week-end = the last 12h candle of each week (its close = end of the last day).
//  Long histories are split across a few log messages so nothing exceeds Pine's
//  log-length limit; each message holds whole lines. Paste them ALL into a sheet
//  and sort by the date column - the order across messages doesn't matter.
// ════════════════════════════════════════════════════════════════════════════
bool   do_export  = input.bool(false, "Export Weekly Equity to Log", group = "Weekly Equity Export")
float  export_val = followed_eq * 10000.0          // scale is arbitrary; returns are scale-free

var string buf    = ""
var float  wk_o   = na                              // running week's open
var float  wk_h   = na                              // running week's high
var float  wk_l   = na                              // running week's low

weekStart        = time("1W")                       // start-of-week stamp; timeframe-independent
bool week_rolled = not na(weekStart[1]) and weekStart != weekStart[1]

// First bar of a new week => the PREVIOUS bar was the prior week's last 12h candle.
// The accumulators still hold the COMPLETED week's O/H/L here (they are reset below),
// and export_val[1] is that week's closing equity.
// Flush the buffer first if adding this line would exceed the safe message length.
if do_export and week_rolled and backtest[1] and not na(wk_o)
    row = str.format_time(time[1], "yyyy-MM-dd", syminfo.timezone) + "\t" + str.tostring(wk_o, "#.##") + "\t" + str.tostring(wk_h, "#.##") + "\t" + str.tostring(wk_l, "#.##") + "\t" + str.tostring(export_val[1], "#.##") + "\n"
    if str.length(buf) + str.length(row) > 3800
        log.info(buf)
        buf := ""
    buf := buf + row

// Accumulate the running week's O/H/L from the per-bar equity. A new week (or the first
// backtested bar) starts fresh from this bar's value; within a week only the extremes move.
if backtest
    if week_rolled or na(wk_o)
        wk_o := export_val
        wk_h := export_val
        wk_l := export_val
    else
        wk_h := math.max(wk_h, export_val)
        wk_l := math.min(wk_l, export_val)

// Final bar: append the current (possibly partial) week, then flush whatever remains.
if do_export and barstate.islast and backtest and not na(wk_o)
    last = str.format_time(time, "yyyy-MM-dd", syminfo.timezone) + "\t" + str.tostring(wk_o, "#.##") + "\t" + str.tostring(wk_h, "#.##") + "\t" + str.tostring(wk_l, "#.##") + "\t" + str.tostring(export_val, "#.##")
    if str.length(buf) + str.length(last) > 3800
        log.info(buf)
        buf := ""
    buf := buf + last
    log.info(buf)
